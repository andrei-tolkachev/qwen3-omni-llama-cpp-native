cmake_minimum_required(VERSION 3.18)
project(lora_omni_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(CMAKE_BUILD_RPATH "$ORIGIN")
if(TARGET llama-server-impl)
    set(LLAMA_SOURCE_DIR "${CMAKE_SOURCE_DIR}")
    set(OMNI_LLAMA llama)
    set(OMNI_GGML ggml)
    set(OMNI_GGML_CPU ggml-cpu)
    set(OMNI_GGML_BASE ggml-base)
    set(OMNI_SERVER llama-server-impl)
    set(OMNI_COMMON llama-common)
    set(OMNI_MTMD mtmd)
else()
    set(LLAMA_SOURCE_DIR "" CACHE PATH "llama.cpp source checkout")
    set(LLAMA_BUILD_DIR "${LLAMA_SOURCE_DIR}/build" CACHE PATH "Matching llama.cpp build")
    if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
        message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the selected llama.cpp checkout")
    endif()
    if(NOT EXISTS "${LLAMA_BUILD_DIR}/CMakeCache.txt")
        message(FATAL_ERROR "LLAMA_BUILD_DIR must be a matching configured llama.cpp build")
    endif()
    file(STRINGS "${LLAMA_BUILD_DIR}/CMakeCache.txt" OMNI_BUILD_SOURCE REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
    string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" OMNI_BUILD_SOURCE "${OMNI_BUILD_SOURCE}")
    get_filename_component(OMNI_BUILD_SOURCE "${OMNI_BUILD_SOURCE}" REALPATH)
    get_filename_component(OMNI_EXPECTED_SOURCE "${LLAMA_SOURCE_DIR}" REALPATH)
    if(NOT OMNI_BUILD_SOURCE STREQUAL OMNI_EXPECTED_SOURCE)
        message(FATAL_ERROR "LLAMA_BUILD_DIR was configured from a different source tree; rebuild a matching library set")
    endif()
    find_library(OMNI_LLAMA llama PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_GGML ggml PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_GGML_CPU ggml-cpu PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_GGML_BASE ggml-base PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_SERVER llama-server-impl PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_COMMON llama-common PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    find_library(OMNI_MTMD mtmd PATHS "${LLAMA_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
    set(CMAKE_BUILD_RPATH "$ORIGIN;${LLAMA_BUILD_DIR}/bin")
endif()
add_library(omni_native STATIC src/backend.cpp src/device_decoder.cpp src/device_codec.cpp src/codec_stream.cpp src/talker.cpp src/predictor.cpp src/projection.cpp src/codec_embedding.cpp src/codec_transformer.cpp src/codec_upsample.cpp src/codec_dac.cpp src/voice.cpp src/duplex.cpp)
target_include_directories(omni_native PUBLIC src "${LLAMA_SOURCE_DIR}/vendor/nlohmann" "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include" PRIVATE vendor "${LLAMA_SOURCE_DIR}/common")
target_link_libraries(omni_native PUBLIC "${OMNI_LLAMA}" "${OMNI_GGML}" "${OMNI_GGML_CPU}" "${OMNI_GGML_BASE}" Threads::Threads)
add_executable(omni-generate src/generate.cpp)
target_link_libraries(omni-generate PRIVATE omni_native)
add_executable(omni-decode src/decode.cpp)
target_link_libraries(omni-decode PRIVATE omni_native)
add_executable(omni-quantize src/quantize.cpp)
target_link_libraries(omni-quantize PRIVATE omni_native)
add_executable(omni-visual-reply-test EXCLUDE_FROM_ALL tests/visual_reply_test.cpp)
target_include_directories(omni-visual-reply-test PRIVATE src "${LLAMA_SOURCE_DIR}/common")
target_link_libraries(omni-visual-reply-test PRIVATE "${OMNI_COMMON}")
target_compile_options(omni-visual-reply-test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG>)
if(UNIX AND NOT APPLE)
    target_link_options(omni-generate PRIVATE "LINKER:--disable-new-dtags")
    target_link_options(omni-decode PRIVATE "LINKER:--disable-new-dtags")
    target_link_options(omni-quantize PRIVATE "LINKER:--disable-new-dtags")
endif()
if(UNIX AND NOT APPLE)
    add_executable(omni-server src/server_adapter.cpp)
    target_include_directories(omni-server PRIVATE "${LLAMA_SOURCE_DIR}/common" "${LLAMA_SOURCE_DIR}/tools/server" "${LLAMA_SOURCE_DIR}/tools/mtmd" "${LLAMA_SOURCE_DIR}/vendor")
    target_link_libraries(omni-server PRIVATE omni_native "${OMNI_SERVER}" "${OMNI_COMMON}" "${OMNI_MTMD}" ${CMAKE_DL_LIBS})
    target_link_options(omni-server PRIVATE "LINKER:--disable-new-dtags")
    set_target_properties(omni-server PROPERTIES ENABLE_EXPORTS ON)
    add_executable(omni-server-security-test EXCLUDE_FROM_ALL tests/server_security_test.cpp)
    target_include_directories(omni-server-security-test PRIVATE src "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include" "${LLAMA_SOURCE_DIR}/common" "${LLAMA_SOURCE_DIR}/tools/server" "${LLAMA_SOURCE_DIR}/vendor")
    target_link_libraries(omni-server-security-test PRIVATE "${OMNI_SERVER}" "${OMNI_COMMON}" "${OMNI_LLAMA}" "${OMNI_GGML_BASE}")
    if(NOT TARGET llama-server-impl)
        # A standalone build links the prebuilt HTTP library without its CMake
        # usage requirements. Match httplib's public class layout: SSL changes
        # ClientImpl even when the test only sends plain HTTP to loopback.
        file(STRINGS "${LLAMA_BUILD_DIR}/CMakeCache.txt" OMNI_HTTP_SSL_OPTIONS
             REGEX "^(LLAMA_OPENSSL|LLAMA_BUILD_BORINGSSL|LLAMA_BUILD_LIBRESSL|OPENSSL_VERSION_SUPPORTED):[^=]*=")
        if(OMNI_HTTP_SSL_OPTIONS MATCHES "LLAMA_BUILD_(BORINGSSL|LIBRESSL):BOOL=(ON|TRUE|1)"
           OR (OMNI_HTTP_SSL_OPTIONS MATCHES "LLAMA_OPENSSL:BOOL=(ON|TRUE|1)"
               AND OMNI_HTTP_SSL_OPTIONS MATCHES "OPENSSL_VERSION_SUPPORTED:INTERNAL=(ON|TRUE|1)"))
            target_compile_definitions(omni-server-security-test PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
        endif()
    endif()
    target_compile_options(omni-server-security-test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG>)
else()
    message(STATUS "The server capture adapter currently requires ELF symbol interposition (Linux); native speech CLI targets remain available")
endif()

add_executable(omni-cache-growth-test EXCLUDE_FROM_ALL tests/cache_growth_test.cpp)
target_link_libraries(omni-cache-growth-test PRIVATE omni_native)
target_compile_options(omni-cache-growth-test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG>)
